.MODEL small
.STACK 100h
.DATA
; ============================================
; 数据段 - 系统消息和变量定义
; ============================================

logo         DB "WELCOME TO LEISURE CLUB SYSTEM", 0Dh, 0Ah, "$"
mainMenu     DB "1. Staff", 0Dh, 0Ah
             DB "2. Member", 0Dh, 0Ah
             DB "3. Exit", 0Dh, 0Ah
             DB "Please Enter Your Selection : $"

; 会员数据数组 - 支持最多 20 个会员
MAX_MEMBERS  EQU 20
memberIDArray        DB MAX_MEMBERS * 5 DUP('$')
memberPasswordArray  DB MAX_MEMBERS * 5 DUP('$')
memberNameArray      DB MAX_MEMBERS * 20 DUP('$')

.CODE
main PROC
    ; 初始化数据段
    mov ax, @data
    mov ds, ax

mainLoop:
    ; 显示欢迎信息
    lea dx, logo
    mov ah, 09h
    int 21h

    ; 显示主菜单
    lea dx, mainMenu
    mov ah, 09h
    int 21h

    ; 读取用户输入
    mov ah, 01h
    int 21h

    ; 根据输入跳转
    cmp al, '1'
    je handleStaff
    cmp al, '2'
    je handleMember
    cmp al, '3'
    jmp exitProgram

; ============================================
; 登录验证 - 比较用户输入与存储的凭据
; ============================================
loginStaff:
    mov ax, 0
    mov al, staffCounter
    mov cx, ax
    mov si, 2    ; 用户输入偏移
    mov di, 0    ; 数组索引

IDValidationL1:
    mov al, inputIDBuffer[si]
    mov bl, staffIDArray[di]
    inc si
    inc di
    cmp bl, "$"
    je loginStaffPassword
    cmp al, bl
    jne IDvalidationL2
    jmp IDvalidationL1
